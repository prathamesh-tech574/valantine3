      "Aww come on! 🥺 It's something really important and special!",
      "Pleeease! 😢 Just listen to what I have to say!",

      "I promise it's worth it! 💫 Say yes?",

    function createStars() {
      const container = document.getElementById('stars-container');
      for (let i = 0; i < 30; i++) {
        const star = document.createElement('div');
        star.className = 'star text-2xl fixed pointer-events-none twinkle';
        star.textContent = ['⭐', '✨', '💫'][Math.floor(Math.random() * 3)];
        star.style.left = Math.random() * 100 + '%';
        star.style.top = Math.random() * 100 + '%';
        star.style.animationDelay = Math.random() * 2 + 's';
        star.style.opacity = Math.random() * 0.5 + 0.3;
        container.appendChild(star);
      }
    }

    function handleInitialNo() {
      initialNoAttempts++;
      const initialNoBtn = document.getElementById('initial-no-btn');
      const initialYesBtn = document.getElementById('initial-yes-btn');
      const counter = document.getElementById('initial-attempt-counter');
      
      initialNoBtn.classList.add('shake-intense');
      setTimeout(() => initialNoBtn.classList.remove('shake-intense'), 800);
      
      const messageIndex = Math.min(initialNoAttempts - 1, initialNoMessages.length - 1);
      counter.classList.remove('hidden');
      counter.textContent = initialNoMessages[messageIndex];
      
      const scale = Math.max(0.6, 1 - (initialNoAttempts * 0.08));
      initialNoBtn.style.transform = `scale(${scale})`;
      initialNoBtn.style.opacity = Math.max(0.4, 1 - (initialNoAttempts * 0.12));
      
      const yesScale = Math.min(1.3, 1 + (initialNoAttempts * 0.06));
      initialYesBtn.style.transform = `scale(${yesScale})`;
      
      if (initialNoAttempts >= 4) {
        const randomX = (Math.random() - 0.5) * 80;
        const randomY = (Math.random() - 0.5) * 40;
        initialNoBtn.style.transform = `scale(${scale}) translate(${randomX}px, ${randomY}px)`;
      }
    }

    function handleInitialYes() {
      currentStep = 'valentine';
      document.getElementById('initial-screen').style.display = 'none';
      document.getElementById('question-screen').classList.remove('hidden');
      createCelebration();
    }

    function handleNo() {
      noAttempts++;
      const noBtn = document.getElementById('no-btn');
      const yesBtn = document.getElementById('yes-btn');
      const definitelyBtn = document.getElementById('definitely-btn');
      const messageBox = document.getElementById('sweet-message');
      const messageText = document.getElementById('message-text');
      const counter = document.getElementById('attempt-counter');
      
      noBtn.classList.add('shake-intense');
      setTimeout(() => noBtn.classList.remove('shake-intense'), 800);
      
      const messageIndex = Math.min(noAttempts - 1, sweetMessages.length - 1);
      messageText.textContent = sweetMessages[messageIndex];
      messageBox.classList.remove('hidden');
      messageBox.classList.add('bounce-scale');
      
      const btnTextIndex = Math.min(noAttempts, noButtonTexts.length - 1);
      noBtn.textContent = noButtonTexts[btnTextIndex];
      
      const scale = Math.max(0.6, 1 - (noAttempts * 0.08));
      const opacity = Math.max(0.4, 1 - (noAttempts * 0.12));
      
      noBtn.style.transform = `scale(${scale})`;
      noBtn.style.opacity = opacity;
      
      const yesScale = Math.min(1.3, 1 + (noAttempts * 0.06));
      yesBtn.style.transform = `scale(${yesScale})`;
      definitelyBtn.style.transform = `scale(${yesScale})`;
      
      if (noAttempts >= 2) {
        counter.classList.remove('hidden');
        counter.textContent = `🎯 ${noAttempts} attempts... but I'm not giving up! 💪`;
      }
      
      if (noAttempts >= 4) {
        const randomX = (Math.random() - 0.5) * 80;
        const randomY = (Math.random() - 0.5) * 40;
        noBtn.style.transform = `scale(${scale}) translate(${randomX}px, ${randomY}px)`;
      }
    }

    function handleYes() {
      if (successAnimationPlayed) return;
      successAnimationPlayed = true;
      
      document.getElementById('question-screen').style.display = 'none';
      document.getElementById('success-screen').classList.remove('hidden');
      
      createCelebration();
      createConfetti();
      setTimeout(createConfetti, 400);
      setTimeout(createConfetti, 800);
      setTimeout(createConfetti, 1200);
    }

    function createCelebration() {
      const celebrationEmojis = ['🎉', '🎊', '💕', '✨', '🌟', '💖', '🎈', '🎁', '💝'];
      
      for (let i = 0; i < 40; i++) {
        setTimeout(() => {
          const emoji = document.createElement('div');
          emoji.textContent = celebrationEmojis[Math.floor(Math.random() * celebrationEmojis.length)];
          emoji.className = 'confetti-piece text-4xl';
          emoji.style.left = Math.random() * 100 + '%';
          emoji.style.top = '-50px';
          emoji.style.setProperty('--tx', (Math.random() - 0.5) * 300 + 'px');
          emoji.style.setProperty('--ty', (Math.random() + 0.5) * 500 + 'px');
          emoji.style.animation = 'confetti-explode 3s ease-out forwards';
          document.body.appendChild(emoji);
          
          setTimeout(() => emoji.remove(), 3000);
        }, i * 50);
      }
    }

    function createConfetti() {
      const colors = ['#667eea', '#764ba2', '#f093fb', '#f5576c', '#ff6b6b'];
      
      for (let i = 0; i < 60; i++) {
        const confetti = document.createElement('div');
        confetti.className = 'confetti-piece';
        confetti.style.width = (Math.random() * 12 + 8) + 'px';
        confetti.style.height = confetti.style.width;
        confetti.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
        confetti.style.borderRadius = Math.random() > 0.5 ? '50%' : '2px';
        confetti.style.left = Math.random() * 100 + '%';
        confetti.style.top = '-10px';
        confetti.style.setProperty('--tx', (Math.random() - 0.5) * 300 + 'px');
        confetti.style.setProperty('--ty', (Math.random() + 0.5) * 600 + 'px');
        confetti.style.animation = 'confetti-explode ' + (Math.random() * 2 + 2.5) + 's ease-in forwards';
        document.body.appendChild(confetti);
        
        setTimeout(() => confetti.remove(), 4000);
      }
    }

    function onConfigChange(config) {
      const initialName = document.getElementById('initial-name');
      const valentineName = document.getElementById('valentine-name');
      const initialQuestion = document.getElementById('initial-question');
      const loveMessageEl = document.getElementById('love-message-text');
      
      if (config.her_name && config.her_name.trim()) {
        initialName.textContent = `Hey ${config.her_name} 👋`;
        valentineName.textContent = `Hey ${config.her_name} 👋`;
      } else {
        initialName.textContent = 'Hey Teresaaa 👋';
        valentineName.textContent = 'Hey Teresaaa 👋';
      }
      
      if (config.initial_question) {
        initialQuestion.textContent = config.initial_question;
      }

      if (config.valentine_message) {
        loveMessageEl.textContent = config.valentine_message;
      }
    }

    function mapToCapabilities(config) {
      return {
        recolorables: [],
        borderables: [],
        fontEditable: undefined,
        fontSizeable: undefined
      };
    }

    function mapToEditPanelValues(config) {
      return new Map([
        ["her_name", config.her_name || defaultConfig.her_name],
        ["initial_question", config.initial_question || defaultConfig.initial_question],
        ["valentine_message", config.valentine_message || defaultConfig.valentine_message]
      ]);
    }

    window.elementSdk.init({
      defaultConfig,
